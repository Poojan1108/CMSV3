// complaintService.js [L53-L72] 
export const getCategoryDepartment = (category) => {
  const cat = (category || '').toLowerCase();
  if (cat.includes('wifi') || cat.includes('lab'))
    return { name: 'IT & Digital Infrastructure' };
  if (cat.includes('plumb') || cat.includes('water'))
    return { name: 'Plumbing & Water Services' };
  if (cat.includes('power') || cat.includes('electric'))
    return { name: 'Electrical & Power Services' };
  return { name: 'Maintenance & Estate Services' };
};
